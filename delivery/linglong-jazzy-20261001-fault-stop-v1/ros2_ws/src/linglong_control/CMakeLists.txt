cmake_minimum_required(VERSION 3.16)
project(linglong_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Allows the actual control core to be compiled/tested without ROS installed.
option(LINGLONG_CORE_ONLY "Build only ROS-independent control tests" OFF)
option(LINGLONG_WITH_IGH "Build the physical four-joint left-arm IgH plugin" OFF)
enable_testing()
add_executable(test_hardware_state test/test_hardware_state.cpp)
target_include_directories(test_hardware_state PRIVATE include)
target_compile_options(test_hardware_state PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME hardware_state COMMAND test_hardware_state)
add_executable(test_left_arm_core test/test_left_arm_core.cpp)
target_include_directories(test_left_arm_core PRIVATE include)
target_compile_options(test_left_arm_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME left_arm_core COMMAND test_left_arm_core)
set(LINGLONG_IGH_TEST_INCLUDE_DIR "" CACHE PATH "Optional official IgH 1.6 header directory for fake-transport tests")
if(LINGLONG_IGH_TEST_INCLUDE_DIR)
  add_executable(test_left_arm_bus test/test_left_arm_bus.cpp)
  target_include_directories(test_left_arm_bus PRIVATE include ${LINGLONG_IGH_TEST_INCLUDE_DIR})
  target_compile_options(test_left_arm_bus PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME left_arm_bus COMMAND test_left_arm_bus)
endif()
if(UNIX)
  find_package(Threads REQUIRED)
  add_executable(sim_validation test/sim_validation.cpp)
  target_include_directories(sim_validation PRIVATE include)
  target_link_libraries(sim_validation Threads::Threads)
  target_compile_options(sim_validation PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(LINGLONG_CORE_ONLY)
  enable_testing()
  add_executable(test_sim_core test/test_sim_core.cpp)
  target_include_directories(test_sim_core PRIVATE include)
  target_compile_options(test_sim_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME sim_core COMMAND test_sim_core)
  add_test(NAME validation_faults COMMAND sim_validation --mode faults --trials 5)
  add_test(NAME validation_tracking COMMAND sim_validation --mode endurance --cycles 10000)
  return()
endif()

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)

add_library(linglong_sim_system SHARED src/sim_system.cpp)
target_include_directories(linglong_sim_system PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_options(linglong_sim_system PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(linglong_sim_system hardware_interface pluginlib rclcpp rclcpp_lifecycle)
pluginlib_export_plugin_description_file(hardware_interface linglong_hardware.xml)
if(LINGLONG_WITH_IGH)
  find_path(IGH_INCLUDE_DIR ecrt.h HINTS /usr/local/etherlab/include /opt/etherlab/include)
  find_library(IGH_LIBRARY ethercat HINTS /usr/local/etherlab/lib /opt/etherlab/lib)
  if(NOT IGH_INCLUDE_DIR OR NOT IGH_LIBRARY)
    message(FATAL_ERROR "LINGLONG_WITH_IGH requires the IgH ecrt.h and libethercat on the controller")
  endif()
  add_library(linglong_left_arm_system SHARED src/left_arm_system.cpp)
  target_include_directories(linglong_left_arm_system PRIVATE include ${IGH_INCLUDE_DIR})
  target_compile_options(linglong_left_arm_system PRIVATE -Wall -Wextra -Wpedantic)
  ament_target_dependencies(linglong_left_arm_system hardware_interface pluginlib rclcpp rclcpp_lifecycle)
  target_link_libraries(linglong_left_arm_system ${IGH_LIBRARY})
  pluginlib_export_plugin_description_file(hardware_interface linglong_left_arm_hardware.xml)
  install(TARGETS linglong_left_arm_system LIBRARY DESTINATION lib)
endif()
install(TARGETS linglong_sim_system EXPORT export_linglong_control
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch urdf DESTINATION share/${PROJECT_NAME})
install(FILES RELEASE_ID DESTINATION share/${PROJECT_NAME})
ament_python_install_package(linglong_control_tools)
install(PROGRAMS scripts/control_diagnostics scripts/trajectory_demo scripts/architecture_probe scripts/fault_snapshot scripts/system_manager scripts/system_transition
  DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(ament_cmake_gtest REQUIRED)
  if(LINGLONG_IGH_TEST_INCLUDE_DIR)
    add_executable(test_left_arm_plugin test/test_left_arm_plugin.cpp src/left_arm_system.cpp)
    target_include_directories(test_left_arm_plugin PRIVATE include ${LINGLONG_IGH_TEST_INCLUDE_DIR})
    target_compile_options(test_left_arm_plugin PRIVATE -Wall -Wextra -Wpedantic)
    ament_target_dependencies(test_left_arm_plugin hardware_interface pluginlib rclcpp rclcpp_lifecycle)
    add_test(NAME left_arm_plugin COMMAND test_left_arm_plugin)
  endif()
  add_executable(test_sim_core test/test_sim_core.cpp)
  target_include_directories(test_sim_core PRIVATE include)
  add_test(NAME sim_core COMMAND test_sim_core)
  ament_add_gtest(test_plugin test/test_plugin.cpp)
  target_link_libraries(test_plugin linglong_sim_system)
  ament_target_dependencies(test_plugin hardware_interface pluginlib rclcpp rclcpp_lifecycle)
  ament_add_pytest_test(test_health test/test_health.py)
  ament_add_pytest_test(test_diagnostics_message test/test_diagnostics_message.py)
  ament_add_pytest_test(test_system_state test/test_system_state.py)
  ament_add_pytest_test(test_system_manager test/test_system_manager.py)
  ament_add_pytest_test(test_acceptance test/test_acceptance.py)
  ament_add_pytest_test(test_evidence test/test_evidence.py)
  ament_add_pytest_test(test_snapshot_callbacks test/test_snapshot_callbacks.py)
  ament_add_pytest_test(test_description test/test_description.py)
  ament_add_pytest_test(test_left_arm_configuration test/test_left_arm_configuration.py)
endif()

ament_export_targets(export_linglong_control HAS_LIBRARY_TARGET)
ament_export_dependencies(hardware_interface pluginlib rclcpp rclcpp_lifecycle)
ament_package()
